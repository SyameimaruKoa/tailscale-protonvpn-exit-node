# Forward only packets arriving on the dedicated Docker network (eth0)
# through the VPN tunnel. Container addresses are auto-assigned by Docker,
# so no fixed source address is used; this network contains only the
# Gluetun and Tailscale containers, and return traffic is limited to
# established/related flows towards eth0.
iptables -A FORWARD -i eth0 -o tun0 -j ACCEPT
iptables -A FORWARD -i tun0 -o eth0 -m conntrack --ctstate ESTABLISHED,RELATED -j ACCEPT
iptables -t nat -A POSTROUTING -o tun0 -j MASQUERADE
ip6tables -A FORWARD -i eth0 -o tun0 -j ACCEPT
ip6tables -A FORWARD -i tun0 -o eth0 -m conntrack --ctstate ESTABLISHED,RELATED -j ACCEPT
ip6tables -t nat -A POSTROUTING -o tun0 -j MASQUERADE
