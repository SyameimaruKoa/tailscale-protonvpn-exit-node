# Allow only the Tailscale gateway's forwarded packets through the VPN tunnel.
iptables -A FORWARD -i eth0 -s 10.2.1.3/32 -o tun0 -j ACCEPT
iptables -A FORWARD -i tun0 -o eth0 -d 10.2.1.3/32 -m conntrack --ctstate ESTABLISHED,RELATED -j ACCEPT
iptables -t nat -A POSTROUTING -s 10.2.1.3/32 -o tun0 -j MASQUERADE
ip6tables -A FORWARD -i eth0 -s fd6a:204b:1e8e::3/128 -o tun0 -j ACCEPT
ip6tables -A FORWARD -i tun0 -o eth0 -d fd6a:204b:1e8e::3/128 -m conntrack --ctstate ESTABLISHED,RELATED -j ACCEPT
ip6tables -t nat -A POSTROUTING -s fd6a:204b:1e8e::3/128 -o tun0 -j MASQUERADE
